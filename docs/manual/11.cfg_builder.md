# 11 CFG 构建

`compiler/codegen/cfg/builder.py` — 将单个 HIR 函数体翻译为 CFG `Function`。

## CfgBuilder

```python
class CfgBuilder:
    def __init__(self, type_ctx, dp: DefPoint, func_name: str)
    def build(self) -> IR.Function
```

### build() 流程

```python
def build(self) -> IR.Function:
    # 1. 创建 entry block
    # 2. 注册 local variables (VarRef)
    # 3. translate HIR block body → CFG stmts
    body_val = self.__translate_block(dp.body)
    # 4. 如果入口 block 未终止, 添加 Ret(body_val)
    # 5. Dead code elimination
    # 6. RPO 排序
    # 7. 终止检查 (__guard_termination)
```

## HIR → CFG 翻译

### __resolve_val — 核心 dispatch

```python
def __resolve_val(self, expr: HIR.Expr) -> IR.Value:
    match expr:
        case HIR.Binary():   return self.__resolve_binary(expr)
        case HIR.Unary():    return self.__resolve_unary(expr)
        case HIR.Call():     return self.__resolve_call(expr)
        case HIR.Var():      return self.__resolve_var(expr)
        case HIR.Let():      return self.__translate_let(expr)
        case HIR.If():       return self.__translate_if(expr)
        case HIR.Loop():     return self.__translate_loop(expr)
        case HIR.Match():    return self.__translate_match(expr)
        ...
```

### 赋值翻译 (__resolve_assign)

```python
def __resolve_assign(self, expr: HIR.Binary) -> IR.Value:
    lhs_addr = self.__resolve_addr(expr.left)   # 目标地址
    rhs_val  = self.__resolve_val(expr.right)   # 源值
    self.__build_store(rhs_val, lhs_addr)       # store value → address
    return rhs_val
```

### 地址解析 (__resolve_addr)

```python
def __resolve_addr(self, expr: HIR.Expr) -> IR.Value:
    if not expr.is_place:
        val = self.__resolve_val(expr)
        return self.__build_alloca(val)        # rvalue → 临时变量
    match expr:
        case HIR.Var():        → var_ptr (取变量地址)
        case HIR.FieldAccess(): → field_ptr (字段偏移)
        case HIR.Unary(Deref):  → 不 load, 直接返回指针值
        case HIR.TupleAccess(): → 元组字段偏移
        case HIR.Ty():          → 函数指针
```

### 变量翻译

```python
def __resolve_var(self, expr: HIR.Var) -> IR.Value:
    addr = self.__resolve_var_addr(expr)    # VarPtr 指令
    return self.__build_load(addr)          # Load 指令

def __build_var_ptr(self, var_ref: IR.VarRef) -> IR.Value:
    result = IR.Reg(type_id=alloc_pointer(var_ref.type_id))
    return IR.VarPtr(result=result, var_ref=var_ref).result
```

## 短路逻辑 (__resolve_logical)

`&&` 和 `||` 通过条件跳转实现短路求值：

```python
def __resolve_logical(self, expr: HIR.Binary) -> IR.Value:
    cond_val = self.__resolve_val(expr.left)
    rhs_block  = self.__new_block("logical.rhs")
    merge_block = self.__new_block("logical.merge")

    if expr.op == LogicalAnd:
        # a && b: if false → short-circuit to false
        self.__set_terminator(CondBr(cond_val, rhs_block, merge_block))
        short_circuit_value = BoolLiteral(False)
    else:
        # a || b: if true → short-circuit to true
        self.__set_terminator(CondBr(cond_val, merge_block, rhs_block))
        short_circuit_value = BoolLiteral(True)

    # rhs block
    self.__switch_to(rhs_block)
    rhs_val = self.__resolve_val(expr.right)
    rhs_end_block = self.__current_block  # 嵌套逻辑后的实际 block
    if rhs_end_block.terminator is None:
        self.__set_terminator(Br(merge_block))

    # merge block
    self.__switch_to(merge_block)
    result = self.__emit_phi([(entry_block, short_circuit_value),
                               (rhs_end_block, rhs_val)])
    cont_block = self.__new_block("logical.cont")
    self.__set_terminator(Br(cont_block))
    self.__switch_to(cont_block)
    return result
```

## Block 管理

```python
def __new_block(self, label: str) -> IR.Block:
    block = IR.Block(f"{label}.{counter}")
    counter += 1
    func.blocks.append(block)
    return block

def __switch_to(self, block: IR.Block):
    self.__current_block = block

def __set_terminator(self, term: IR.Terminator):
    self.__current_block.terminator = term
```

## 终止检查 (__guard_termination)

```python
def __guard_termination(self, dp: DefPoint):
    for block in func.blocks:
        if block.terminator is None:
            if ret_ty == void:
                block.terminator = Ret(void_reg)    # 自动补 Ret
            else:
                raise CodegenError("unterminated block")  # 非 void 必须显式 return
```

## 正则化

- **Dead code elimination**：BFS 从 entry block 开始，移除不可达 block，清理 phi 节点中来自已删除 block 的条目
- **RPO 排序**：按逆后序排列 block（`__sort_blocks_rpo`），保证 phi 节点的前驱 block 在 phi 之前定义

## CfgTranslator

`compiler/codegen/cfg/translator.py` — 多函数调度器。

```python
class CfgTranslator:
    def run(self, def_points: dict[int, DefPoint]):
        for type_id, dp in def_points.items():
            func = self.__translate(dp)           # CfgBuilder.build()
            self.__functions[type_id] = func
```
